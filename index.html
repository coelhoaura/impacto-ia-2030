Query Engine 
-- Desafio 1: Diagnostico de Elevacao de Escolaridade
WITH Vagas_Escaladas AS (
  SELECT
    s.industry,
    COUNT(f.job_record_id) AS total_vagas_escaladas
  FROM fato_ai_job_impact f
  JOIN dim_setor s ON f.setor_id = s.setor_id
  WHERE f.education_shift_type = 'Exigencia Elevada (Upgraded)'
  GROUP BY s.industry
)
SELECT
  v.industry,
  v.total_vagas_escaladas,
  SUM(v.total_vagas_escaladas) OVER () AS total_geral,
  ROUND(v.total_vagas_escaladas * 100.0 / SUM(v.total_vagas_escaladas) OVER (), 2) AS pct_representatividade
FROM Vagas_Escaladas v
ORDER BY pct_representatividade DESC;

-- Desafio 2 & 3: Ranking e Quartis (NTILE) de Exposicao IA
SELECT
  c.job_title,
  f.salary_growth_pct,
  NTILE(4) OVER (ORDER BY f.ai_exposure_index DESC) AS exposure_quartile,
  ROW_NUMBER() OVER (ORDER BY f.salary_growth_pct DESC) AS rank_vencedores
FROM fato_ai_job_impact f
JOIN dim_cargo c ON f.cargo_id = c.cargo_id;

DAX Measures 
// Medida 1: Saldo Liquido de Vagas (Volume)
Saldo_Vagas_Liquido_k =
SUM('fato_ai_job_impact'[Job_Volume_2030_k]) -
SUM('fato_ai_job_impact'[Job_Volume_Pre_AI_k])

// Medida 2: Variacao Salarial Percentual
Variacao_Salarial_% =
DIVIDE(
    AVERAGE('fato_ai_job_impact'[Salary_Post_AI_USD]) -
    AVERAGE('fato_ai_job_impact'[Salary_Pre_AI_USD]),
    AVERAGE('fato_ai_job_impact'[Salary_Pre_AI_USD]),
    0
)

// Medida 3: Taxa de Elevacao de Escolaridade (Degree Escalation)
Taxa_Elevacao_Escolaridade =
DIVIDE(
    CALCULATE(
        COUNTROWS('fato_ai_job_impact'),
        'fato_ai_job_impact'[Education_Shift_Type] = "Exigencia Elevada (Upgraded)"
    ),
    COUNTROWS('fato_ai_job_impact'),
    0
)

// Medida 4: Formatacao Condicional Dinamica (Cores)
Cor_Variacao_Salarial =
SWITCH(
    TRUE(),
    [Variacao_Salarial_%] > 0.15, "#10B981",
    [Variacao_Salarial_%] >= 0, "#0EA5E9",
    "#EF4444"
)
